rules_version = '2';

// Renni Command Center Firestore rules.
// Approval truth lives here — not in linked docs.
//
// Role model (users/{uid}.role):
//   admin   - instructor / program lead; may override anything
//   coceo   - Co-CEO; may approve any deliverable, override due dates, edit roster-linked data
//   coo|cfo|cmo|csgo - department chiefs
//   member  - student, department-scoped
//
// Deliverable statuses: draft | in_review | needs_revision | approved
// Owners move draft <-> in_review. Approvers route in_review -> approved | needs_revision.
// "blocked" is a task-level flag, not a review outcome.

service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function userDoc() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function hasRole(r) {
      return isSignedIn() && userDoc().role == r;
    }

    function isAdmin() { return hasRole('admin'); }
    function isCoCEO() { return hasRole('coceo'); }
    function isAdminOrCoCEO() { return isAdmin() || isCoCEO(); }
    // Derive chief access from the current role, not the stored isChief
    // boolean. The boolean is roster/profile display metadata and can drift
    // during a role correction; permissions must follow the role immediately.
    function isChief() {
      return isSignedIn()
        && userDoc().role in ['coceo', 'coo', 'cfo', 'cmo', 'csgo', 'admin'];
    }
    // Department-chief approval helper: the user's CURRENT role is in the
    // chief department set (excluding admin/coceo, which are global) AND
    // the deliverable's department matches the user's department.
    // Mirrors app/utils/approvalPermissions.ts so the UI gate and the
    // backend gate stay in lockstep.
    function isDepartmentChiefForDeliverable() {
      return isSignedIn()
        && userDoc().role in ['coo', 'cfo', 'cmo', 'csgo']
        && resource.data.department == userDoc().department;
    }
    function deliverableDoc(id) {
      return get(/databases/$(database)/documents/deliverables/$(id)).data;
    }
    function hasDeliverableDoc(id) {
      return exists(/databases/$(database)/documents/deliverables/$(id));
    }
    function deliverableOutputEditable(id) {
      return hasDeliverableDoc(id)
        && deliverableDoc(id).status in ['draft', 'needs_revision'];
    }
    function isDepartmentChiefForDepartment(dept) {
      return isSignedIn()
        && userDoc().role in ['coo', 'cfo', 'cmo', 'csgo']
        && userDoc().department == dept;
    }
    function teamPulseCycleDoc(cycleId) {
      return get(/databases/$(database)/documents/teamPulseCycles/$(cycleId)).data;
    }
    function teamPulseCycleIsOpen(cycleId) {
      return exists(/databases/$(database)/documents/teamPulseCycles/$(cycleId))
        && teamPulseCycleDoc(cycleId).status == 'open';
    }
    function teamPulseCycleIsClosed(cycleId) {
      return exists(/databases/$(database)/documents/teamPulseCycles/$(cycleId))
        && teamPulseCycleDoc(cycleId).status in ['closed', 'summarized'];
    }

    // -------- roster --------
    // Seed-populated registry of allowed accounts, keyed by email.
    // Read-only to any authed user; writes are admin-only (seed scripts use the admin SDK).
    match /roster/{email} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    // -------- users --------
    // Self-read allowed; writes are admin-only — the /api/auth/provision server route
    // uses the admin SDK to upsert users/{uid} from the roster on first sign-in.
    match /users/{userId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    // -------- deliverables --------
    // Deliverable status is strictly: draft | in_review | needs_revision | approved.
    // `blocked` is task-only and must never land here. There is no blanket
    // admin/coceo update branch — instructor overrides go through the same
    // narrowed branches below so status can only move along legal edges and
    // arbitrary fields can't be rewritten.
    //
    // `statusHistory` is an append-only activity feed written by the client.
    // It is allowed in every branch that mutates the deliverable so the UI
    // can log the event alongside the state change, but the rules do not
    // attempt to make it tamper-proof — that belongs to a server-side
    // lifecycle endpoint (future hardening).
    match /deliverables/{id} {
      allow read: if isSignedIn();
      allow create, delete: if isAdminOrCoCEO();

      allow update: if
        // Owner: save notes/linkedDocUrl or submit (draft|needs_revision → in_review).
        (
          resource.data.ownerUid == request.auth.uid
          && resource.data.status in ['draft', 'needs_revision']
          && request.resource.data.status in ['draft', 'needs_revision', 'in_review']
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['status', 'notes', 'linkedDocUrl', 'submittedForReviewAt',
                'submittedByUid', 'submittedByEmail', 'submittedAt',
                'statusHistory', 'updatedAt']
             )
        )
        // Approver (or admin/coceo override, or matching-department chief):
        // return in_review → needs_revision. Must supply a non-empty
        // returnedReason so the owner knows what to fix.
        || (
          (
            resource.data.approverUid == request.auth.uid
            || isAdminOrCoCEO()
            || isDepartmentChiefForDeliverable()
          )
          && resource.data.status == 'in_review'
          && request.resource.data.status == 'needs_revision'
          && request.resource.data.returnedReason is string
          && request.resource.data.returnedReason.size() > 0
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['status', 'returnedReason', 'reviewedByUid',
                'reviewedByEmail', 'reviewedAt', 'statusHistory', 'updatedAt']
             )
        )
        // Approver (or admin/coceo override, or matching-department chief):
        // approve in_review → approved. Required audit fields must be
        // populated by the client.
        || (
          (
            resource.data.approverUid == request.auth.uid
            || isAdminOrCoCEO()
            || isDepartmentChiefForDeliverable()
          )
          && resource.data.status == 'in_review'
          && request.resource.data.status == 'approved'
          && request.resource.data.approvedAt is string
          && request.resource.data.approvedByUid is string
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['status', 'approvalNotes', 'approvedAt', 'approvedByUid',
                'approvedByEmail', 'approvedByRole', 'reviewedByUid',
                'reviewedByEmail', 'reviewedAt', 'statusHistory', 'updatedAt']
             )
        )
        // Admin/coceo: edit due-date fields only; must not change status.
        || (
          isAdminOrCoCEO()
          && request.resource.data.status == resource.data.status
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['dueDate', 'lastDueDateEditedBy', 'lastDueDateEditedAt',
                'dueDateOverrideReason', 'statusHistory', 'updatedAt']
             )
        );
    }

    // -------- tasks --------
    // Tightened for the Timeline Planner: task creation and planning-field
    // updates are limited to admin / Co-CEO / any chief. Task owners keep
    // the ability to update their own record (status, blocked reason, notes)
    // so /tasks still works for members.
    match /tasks/{id} {
      allow read: if isSignedIn();
      allow create: if isAdmin() || isCoCEO() || isChief();
      allow update: if isAdmin() || isCoCEO() || isChief()
        || resource.data.ownerUid == request.auth.uid;
      allow delete: if isAdminOrCoCEO();
    }

    // -------- goals --------
    match /goals/{id} {
      allow read: if isSignedIn();
      allow create: if isAdminOrCoCEO();
      allow update: if isAdminOrCoCEO()
        || (isChief() && resource.data.ownerUid == request.auth.uid);
      allow delete: if isAdmin();
    }

    // -------- bmc --------
    // All authed users can read. Writes restricted to chiefs, Co-CEOs, and admin.
    // Section-level soft locking is enforced in app code on top of this.
    match /bmc/{id} {
      allow read: if isSignedIn();
      allow write: if isChief() || isCoCEO() || isAdmin();
    }

    // -------- bmcBlocks --------
    // One doc per Business Model Canvas block (customer_segments, etc.).
    // Read: any signed-in user. Edit: admin, Co-CEO, or CSGO. The block's
    // ownerDepartment chief isn't granted write here to keep V1 simple —
    // expand later if granular ownership is worth the rule cost.
    match /bmcBlocks/{id} {
      allow read: if isSignedIn();
      allow create, update, delete:
        if isAdmin() || isCoCEO() || hasRole('csgo');
    }

    // -------- continuity --------
    match /continuity/{id} {
      allow read: if isSignedIn();
      allow write: if isChief() || isCoCEO() || isAdmin();
    }

    // -------- decisions --------
    match /decisions/{id} {
      allow read: if isSignedIn();
      allow create, update: if isChief() || isAdmin();
      allow delete: if isAdmin();
    }

    // -------- deliverable outputs --------
    // Student-authored artifact content, keyed by deliverable id.
    // Read: any signed-in user — chiefs and approvers need to see what
    //   owners have written, the same way they see the deliverable itself.
    // Write: any rostered class member (admin / Co-CEO / COO / chief /
    //   member / owner) when the deliverable is in an editable state.
    //   Approval is a separate axis (handled on /deliverables) — being
    //   able to author output never grants approval rights, due-date
    //   edit rights, or admin powers. The status-based read-only lock
    //   on in_review / approved is preserved by deliverableOutputEditable.
    //
    // Soft Section Locking sprint: writes that touch the `sectionLocks`
    // map flow through this same `allow create, update` branch — locks
    // never bypass deliverableOutputEditable, so a user who can't
    // author the deliverable can't acquire a lock on its sections, and
    // approval/due-date semantics are unchanged.
    //
    // Stronger lock-ownership guarantees (lockedBy must equal
    // request.auth.uid; expired locks may be taken over; only the
    // owner of a lock may release it) are enforced in app code in
    // app/composables/useDeliverableOutputs.ts. CEL has no map-iteration
    // primitives that would let us encode "for each new sectionLocks
    // entry, lockedBy must equal auth.uid" without listing every
    // section id, so we follow the same pattern the bmc rule uses
    // ("Section-level soft locking is enforced in app code on top of
    // this") and rely on the writeable predicate above plus the
    // updatedByUid / updatedByEmail audit fields on the parent doc
    // for traceability.
    match /deliverableOutputs/{deliverableId} {
      allow read: if isSignedIn();
      allow create, update: if deliverableOutputEditable(deliverableId)
        && (
          isAdmin() || isCoCEO() || hasRole('coo')
          || deliverableDoc(deliverableId).ownerUid == request.auth.uid
          || (isChief()
            && deliverableDoc(deliverableId).department == userDoc().department)
          // V2 collaborative drafting: any rostered member can author
          // section content while status is draft / needs_revision.
          // Approval, due-date edit, and admin powers remain untouched.
          || hasRole('member')
        );
      allow delete: if isAdminOrCoCEO() && deliverableOutputEditable(deliverableId);
    }

    // -------- pricing scenarios --------
    // Anyone signed-in can read. Mutations are limited to admin/Co-CEO/CFO
    // since pricing decisions are financial and the CFO owns the model.
    // Narrower than the previous "any signed-in user can create/update"
    // so members/chiefs outside finance can't change numbers.
    match /pricingScenarios/{id} {
      allow read: if isSignedIn();
      allow create, update, delete: if isAdmin() || isCoCEO() || hasRole('cfo');
    }

    // -------- donations --------
    match /donations/{id} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isAdminOrCoCEO()
        || resource.data.recordedByUid == request.auth.uid;
      allow delete: if isAdminOrCoCEO();
    }

    // -------- transactions --------
    match /transactions/{id} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update, delete: if isAdminOrCoCEO();
    }

    // -------- popUpTransactions --------
    // Unified pop-up ledger: sale | donation | adjustment.
    // Students may record sales/donations during the pop-up, but only
    // admin / Co-CEO / CFO can edit or delete historical records so the
    // ledger stays append-mostly and auditable.
    match /popUpTransactions/{id} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update, delete: if isAdmin() || isCoCEO() || hasRole('cfo');
    }

    // -------- aiAdvisorInvocations --------
    // Server-side audit log for Executive Advisor calls. Written
    // exclusively by the /api/ai/executive-advisor endpoint via the
    // Firebase Admin SDK; clients have no read or write surface in
    // V1. Admin / instructor reads happen through the Firebase
    // Console or future server-side admin tooling. Chiefs do NOT
    // get broad read access in V1 — when chief read is added later
    // it must be scoped to the chief's own invocation metadata
    // only (per the Pass 1 brief).
    //
    // Stored fields are minimal metadata only. The endpoint never
    // writes the full prompt, the raw provider response, the
    // Firebase ID token, unrelated student writing, or private
    // profile data into this collection. See
    // server/utils/executiveAdvisor/aiAdvisorInvocations.ts.
    match /aiAdvisorInvocations/{invocationId} {
      allow read, write: if false;
    }

    // -------- aiReviewCoachingInvocations --------
    // Server-side audit log for AI Leadership Coaching calls. Written
    // exclusively by the /api/ai/review-coaching endpoint via the
    // Firebase Admin SDK; clients have no read or write surface.
    // Admin / instructor reads happen through the Firebase Console
    // or future server-side admin tooling. The endpoint never
    // writes the deterministic payload, the AI response body, the
    // raw provider response, the Firebase ID token, full student
    // section excerpts, or private profile data into this
    // collection. See server/utils/aiReviewCoachingInvocations.ts.
    match /aiReviewCoachingInvocations/{invocationId} {
      allow read, write: if false;
    }

    // -------- aiReviewSnapshots --------
    // Metadata-only saved report snapshots. Written and read only
    // through server endpoints so ordinary clients cannot enumerate
    // leadership review history or write their own snapshots.
    match /aiReviewSnapshots/{snapshotId} {
      allow read, write: if false;
    }

    // -------- Team Pulse 1: Calibration Check-In --------
    // Calibration and coaching only. Not an approval workflow, not a
    // leaderboard, and not an automatic grading surface.
    match /teamPulseCycles/{cycleId} {
      allow read: if isSignedIn();
      allow create, update, delete: if isAdmin();
    }

    match /teamPulseResponses/{responseId} {
      allow read: if isAdmin()
        || (isSignedIn() && resource.data.raterUid == request.auth.uid);
      allow create: if isSignedIn()
        && request.resource.data.raterUid == request.auth.uid
        && teamPulseCycleIsOpen(request.resource.data.cycleId);
      allow update: if isSignedIn()
        && resource.data.raterUid == request.auth.uid
        && request.resource.data.raterUid == resource.data.raterUid
        && request.resource.data.rateeUid == resource.data.rateeUid
        && request.resource.data.cycleId == resource.data.cycleId
        && teamPulseCycleIsOpen(resource.data.cycleId);
      allow delete: if isAdmin();
    }

    match /teamPulseSummaries/{summaryId} {
      allow read: if isAdmin()
        || (
          isSignedIn()
          && resource.data.scope in ['company', 'department']
          && (isCoCEO() || hasRole('coo'))
        )
        || (
          isSignedIn()
          && resource.data.scope == 'department'
          && isDepartmentChiefForDepartment(resource.data.subjectDepartment)
        )
        || (
          isSignedIn()
          && resource.data.scope == 'student'
          && resource.data.subjectUid == request.auth.uid
          && teamPulseCycleIsClosed(resource.data.cycleId)
        );
      allow create, update, delete: if isAdmin();
    }

    // -------- feedback --------
    match /feedback/{id} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isAdminOrCoCEO()
        || resource.data.capturedByUid == request.auth.uid;
      allow delete: if isAdmin();
    }

    // ============================================================
    // Our City Studio
    //
    // Deliberately independent permission system from the Renni
    // Command Center rules above. Studio identity lives in
    // `studentProfiles/{uid}` (doc id == Firebase Auth uid), never
    // in `users/{uid}`. A Renni role (admin/coceo/etc.) grants
    // nothing here; only a `studentProfiles` doc with a matching
    // studioRole does. See app/types/studio/models.ts and
    // server/api/studio/auth/provision.post.ts.
    //
    // Only Core entities (08_DATA_MODEL.md) are covered so far —
    // Game entities (gameSessions, properties, studentBusinesses,
    // etc.) get their own rules when Rose City Marketplace / Market
    // Day are built (Phase 3/4).
    // ============================================================

    function studentProfileDoc() {
      return get(/databases/$(database)/documents/studentProfiles/$(request.auth.uid)).data;
    }
    function isStudioEnrolled() {
      return isSignedIn()
        && exists(/databases/$(database)/documents/studentProfiles/$(request.auth.uid));
    }
    function studioRoleOf() {
      return studentProfileDoc().studioRole;
    }
    function isStudioStudent() {
      return isStudioEnrolled() && studioRoleOf() == 'student';
    }
    function isStudioCoach() {
      return isStudioEnrolled() && studioRoleOf() == 'coach';
    }
    function isStudioAdmin() {
      return isStudioEnrolled() && studioRoleOf() == 'admin';
    }
    function isStudioCoachOrAdmin() {
      return isStudioCoach() || isStudioAdmin();
    }
    function isOwnStudioDoc(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }
    // A coach only sees students who share at least one cohort with
    // them — never the whole program.
    function targetStudentCohortIds(studentUid) {
      return get(/databases/$(database)/documents/studentProfiles/$(studentUid)).data.studioCohortIds;
    }
    function isCoachForStudent(studentUid) {
      return isStudioCoach()
        && targetStudentCohortIds(studentUid).hasAny(studentProfileDoc().studioCohortIds);
    }

    // -------- studioRoster --------
    // Seed-populated enrollment registry, mirrors /roster. Only
    // coach/admin need to read it (enrollment lookup UI); students
    // never do. Provisioning itself runs server-side via the Admin
    // SDK and bypasses these rules.
    match /studioRoster/{email} {
      allow read: if isStudioCoachOrAdmin();
      allow write: if isStudioAdmin();
    }

    // -------- studentProfiles --------
    match /studentProfiles/{uid} {
      allow read: if isOwnStudioDoc(uid) || isStudioAdmin() || isCoachForStudent(uid);
      // Self-serve edits are limited to non-privileged fields.
      // studioRole and studioCohortIds are enrollment decisions and
      // change only through admin action — a privilege-escalation
      // guard, since a student rewriting their own studioRole to
      // 'admin' must never be possible via a client write.
      allow update: if isStudioAdmin()
        || (
          isOwnStudioDoc(uid)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['gradeLevel', 'school', 'worksite', 'interests', 'goals',
                'profileStatus', 'activeStudioCohortId', 'updatedAt']
             )
          && request.resource.data.activeStudioCohortId in resource.data.studioCohortIds
        );
      // Created only by the provisioning endpoint via the Admin SDK
      // (which bypasses rules entirely) — no client-side create.
      allow create, delete: if false;
    }

    // -------- cohorts --------
    // Settings (takeHomePayFactor, landlordIncomeMultiplier, etc.)
    // are admin-adjustable only, per the confirmed build decision.
    match /cohorts/{cohortId} {
      allow read: if isStudioAdmin()
        || (isStudioEnrolled() && studentProfileDoc().studioCohortIds.hasAny([cohortId]));
      allow write: if isStudioAdmin();
    }

    // -------- missions --------
    // Seeded content, admin-editable, readable by anyone enrolled.
    match /missions/{id} {
      allow read: if isStudioEnrolled();
      allow write: if isStudioAdmin();
    }

    // -------- studentMissionProgress --------
    match /studentMissionProgress/{id} {
      allow read: if isSignedIn() && (
        resource.data.studentUid == request.auth.uid
        || isStudioAdmin()
        || isCoachForStudent(resource.data.studentUid)
      );
      allow create: if isSignedIn() && request.resource.data.studentUid == request.auth.uid;
      allow update: if isStudioAdmin()
        || (
          isSignedIn() && resource.data.studentUid == request.auth.uid
          && request.resource.data.studentUid == resource.data.studentUid
          && request.resource.data.missionId == resource.data.missionId
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['status', 'startedAt', 'submittedAt', 'updatedAt']
             )
        )
        || (
          isCoachForStudent(resource.data.studentUid)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['coachFeedback', 'reviewedAt', 'status', 'updatedAt']
             )
        );
      allow delete: if isStudioAdmin();
    }

    // -------- worksheetResponses --------
    match /worksheetResponses/{id} {
      allow read: if isSignedIn() && (
        resource.data.studentUid == request.auth.uid
        || isStudioAdmin()
        || isCoachForStudent(resource.data.studentUid)
      );
      allow create: if isSignedIn() && request.resource.data.studentUid == request.auth.uid;
      allow update, delete: if isStudioAdmin()
        || (isSignedIn() && resource.data.studentUid == request.auth.uid);
    }

    // -------- portfolioArtifacts --------
    match /portfolioArtifacts/{id} {
      allow read: if isSignedIn() && (
        resource.data.studentUid == request.auth.uid
        || isStudioAdmin()
        || isCoachForStudent(resource.data.studentUid)
      );
      allow create: if isSignedIn()
        && request.resource.data.studentUid == request.auth.uid
        && request.resource.data.coachStatus == 'draft';
      allow update: if isStudioAdmin()
        || (
          isSignedIn() && resource.data.studentUid == request.auth.uid
          && request.resource.data.studentUid == resource.data.studentUid
          && resource.data.coachStatus in ['draft', 'needs_review']
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['title', 'content', 'coachStatus', 'exportedAt', 'version', 'updatedAt']
             )
        )
        || (
          isCoachForStudent(resource.data.studentUid)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
               ['coachStatus', 'coachFeedback', 'updatedAt']
             )
        );
      allow delete: if isStudioAdmin()
        || (
          isSignedIn() && resource.data.studentUid == request.auth.uid
          && resource.data.coachStatus == 'draft'
        );
    }

    // -------- aiSessions --------
    // Audit log written by the /api/studio/ai/sherpa endpoint via
    // the Admin SDK. No client write path.
    match /aiSessions/{id} {
      allow read: if isSignedIn() && (
        resource.data.studentUid == request.auth.uid
        || isStudioAdmin()
        || isCoachForStudent(resource.data.studentUid)
      );
      allow write: if false;
    }
  }
}
